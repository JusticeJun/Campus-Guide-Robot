cmake_minimum_required(VERSION 3.8)
project(ackermann_rpp)
find_package(ament_cmake REQUIRED)
set(deps nav2_regulated_pure_pursuit_controller nav2_core nav2_util nav2_costmap_2d pluginlib rclcpp rclcpp_lifecycle geometry_msgs nav_msgs tf2 tf2_ros)
foreach(dep IN LISTS deps)
  find_package(${dep} REQUIRED)
endforeach()
add_library(${PROJECT_NAME} SHARED src/controller.cpp)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
ament_target_dependencies(${PROJECT_NAME} ${deps})
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
if(BUILD_TESTING)
 find_package(ament_cmake_gtest REQUIRED)
 ament_add_gtest(test_controller test/test_controller.cpp)
 ament_target_dependencies(test_controller ${deps})
endif()
ament_package()
