cmake_minimum_required(VERSION 3.8)
project(localization_recovery_bt)
find_package(ament_cmake REQUIRED)
find_package(behaviortree_cpp_v3 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
add_library(${PROJECT_NAME} SHARED src/localization_ready.cpp)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
target_compile_definitions(${PROJECT_NAME} PRIVATE BT_PLUGIN_EXPORT)
ament_target_dependencies(${PROJECT_NAME} behaviortree_cpp_v3 rclcpp std_msgs)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_localization_ready test/test_localization_ready.cpp)
  ament_target_dependencies(test_localization_ready behaviortree_cpp_v3 rclcpp std_msgs)
endif()
ament_package()
